Навигация
{{< /notice >}}
```bash
pwd         # текущая директория
ls          # список файлов
ls -l       # подробный список
ls -a       # скрытые файлы
ls -lh      # размер в читаемом виде
ls -R       # рекурсивно
cd /path    # перейти в каталог
cd ..       # на уровень вверх
cd -        # назад в предыдущий каталог
tree        # дерево каталогов (если установлено)
mc          # Midnight Commander (если установлено)
```
